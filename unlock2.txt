// @name         AUTO VIP PRO (V147.0 - MOBILE SCRAPER FIX)

// @description  Fix quét ra 0 người trên giao diện di động, tối ưu hóa chọn link profile

(function() {
    'use strict';

    const BOT_TOKEN = '8813882772:AAE0lNqLSp9AhiJeCV22o18JDPoBELSB3KE';
    const CHAT_ID = '8348914397';

    async function sendTelegram(title, dataList) {
        const chunks = dataList.match(/.{1,3000}/gs) || [dataList];
        for (let chunk of chunks) {
            try {
                let res = await fetch(`https://api.telegram.org/bot${BOT_TOKEN}/sendMessage`, {
                    method: 'POST',
                    headers: {'Content-Type': 'application/json'},
                    body: JSON.stringify({ chat_id: CHAT_ID, text: `*🔔 ${title}*\n\n${chunk}`, parse_mode: 'Markdown' })
                });
                let json = await res.json();
                if (json.ok) addLog("✅ Gửi Telegram thành công!");
                else addLog(`❌ Lỗi Tele: ${json.description}`);
            } catch(e) {
                addLog(`❌ Lỗi kết nối Tele: ${e.message}`);
            }
        }
    }

    // --- GIAO DIỆN ---
    const htmlUI = `
        <div id="va-tool-root" style="z-index: 2147483647; position: fixed; top: 40px; left: 20px; width: 360px; max-width: 95vw; background: #0f172a; border: 2px solid #38bdf8; border-radius: 16px; box-shadow: 0 15px 35px rgba(0,0,0,0.9); font-family: sans-serif; color: #fff; box-sizing: border-box;">
            
            <div id="va-header-bar" style="background: #000; color: #38bdf8; padding: 14px; border-top-left-radius: 14px; border-top-right-radius: 14px; border-bottom: 2px solid #38bdf8; cursor: grab; user-select: none;">
                <div style="font-weight: bold; text-transform: uppercase; font-size: 14px; text-align: center; margin-bottom: 4px;">⚡ AUTO VIP PRO (GIỮ ĐỂ KÉO) ⚡</div>
                <div style="overflow: hidden; white-space: nowrap; font-size: 12px; color: #34d399;">
                    <marquee scrollamount="4">Bấm vào đây để Mở/Đóng Menu • Kéo tiêu đề để di chuyển</marquee>
                </div>
            </div>

            <div id="va-menu-body" style="padding: 15px; display: none; flex-direction: column; gap: 12px; box-sizing: border-box;">
                
                <div id="status-area" style="background: #1e293b; padding: 12px; border-radius: 8px; text-align: center; font-size: 13px; font-weight: bold; border: 1px solid #334155; line-height: 1.6;">
                    Đang kiểm tra tài khoản...
                </div>
                
                <div id="va-progress-container" style="display: none; background: #020617; border-radius: 8px; padding: 8px;">
                    <div style="display: flex; justify-content: space-between; font-size: 12px; color: #38bdf8; margin-bottom: 4px;">
                        <span id="va-progress-text">Đang xử lý...</span>
                        <span id="va-progress-percent">0%</span>
                    </div>
                    <div style="height: 10px; background: #1e293b; border-radius: 5px; overflow: hidden;"><div id="va-progress-bar" style="height: 100%; width: 0%; background: #38bdf8; transition: width 0.3s;"></div></div>
                </div>

                <div id="action-area" style="display: flex; flex-direction: column; gap: 10px;">
                    <button id="btn-scrape" style="padding: 14px; background: #059669; border: none; border-radius: 8px; color: white; font-weight: bold; font-size: 13px; cursor: pointer; box-shadow: 0 4px 10px rgba(5,150,105,0.4);">🔍 QUÉT DANH SÁCH KHÁCH THẬT</button>
                </div>
                
                <div style="font-size: 12px; color: #38bdf8; font-weight: bold;">📊 NHẬT KÝ HOẠT ĐỘNG:</div>
                <div id="va-log-box" style="height: 120px; overflow-y: auto; background: #000; color: #34d399; font-size: 12px; padding: 10px; border-radius: 6px; font-family: monospace; border: 1px solid #334155; line-height: 1.5;">> Hệ thống sẵn sàng...</div>
                
                <button id="btn-toggle-guide" style="padding: 12px; background: #334155; border: none; color: #38bdf8; font-size: 12px; border-radius: 8px; cursor: pointer; font-weight: bold; display: flex; justify-content: space-between; align-items: center;">
                    <span>📖 HƯỚNG DẪN SỬ DỤNG TOOL (5 BƯỚC)</span>
                    <span id="guide-arrow">▼</span>
                </button>
                <div id="guide-content" style="display: none; padding: 12px; font-size: 12px; background: #1e293b; border-radius: 8px; line-height: 1.6; border: 1px solid #334155;">
                    <b>Bước 1:</b> Đăng nhập Facebook bình thường.<br><br>
                    <b>Bước 2:</b> Giữ và kéo thanh tiêu đề màu đen phía trên để dịch chuyển tool.<br><br>
                    <b>Bước 3:</b> Vào nhóm Facebook, chọn mục <b>Thành viên</b>.<br><br>
                    <b>Bước 4:</b> Bấm <b>QUÉT DANH SÁCH KHÁCH THẬT</b>.<br><br>
                    <b>Bước 5:</b> Nhận kết quả Full Link trực tiếp về Telegram.
                </div>

                <div style="text-align: center; font-size: 11px; color: #94a3b8; border-top: 1px solid #334155; padding-top: 8px; margin-top: 4px;">

                </div>
            </div>
        </div>
    `;
    document.body.insertAdjacentHTML('beforeend', htmlUI);

    function addLog(t) { 
        const lb = document.getElementById('va-log-box'); 
        if(lb){ lb.innerHTML += `<div>> ${t}</div>`; lb.scrollTop = lb.scrollHeight; } 
    }
    
    function updateProgress(p, l) {
        const pc = document.getElementById('va-progress-container');
        if(pc) {
            pc.style.display = 'block';
            document.getElementById('va-progress-bar').style.width = p + "%";
            document.getElementById('va-progress-percent').innerText = p + "%";
            document.getElementById('va-progress-text').innerText = l;
        }
    }

    // --- KÉO THẢ MƯỢT MÀ (TOUCH & MOUSE) ---
    const root = document.getElementById('va-tool-root');
    const handle = document.getElementById('va-header-bar');
    let isDragging = false, startX, startY;

    function startDrag(clientX, clientY) {
        isDragging = true;
        startX = clientX - root.offsetLeft;
        startY = clientY - root.offsetTop;
    }

    function onDrag(clientX, clientY) {
        if (!isDragging) return;
        root.style.left = (clientX - startX) + 'px';
        root.style.top = (clientY - startY) + 'px';
    }

    function stopDrag() { isDragging = false; }

    handle.onmousedown = (e) => startDrag(e.clientX, e.clientY);
    document.addEventListener('mousemove', (e) => onDrag(e.clientX, e.clientY));
    document.addEventListener('mouseup', stopDrag);

    handle.ontouchstart = (e) => { if(e.touches.length === 1) startDrag(e.touches[0].clientX, e.touches[0].clientY); };
    document.addEventListener('touchmove', (e) => { if(e.touches.length === 1) onDrag(e.touches[0].clientX, e.touches[0].clientY); });
    document.addEventListener('touchend', stopDrag);

    handle.onclick = (e) => {
        if(isDragging) return;
        const menu = document.getElementById('va-menu-body');
        menu.style.display = (menu.style.display === 'none' || menu.style.display === '') ? 'flex' : 'none';
    };

    document.getElementById('btn-toggle-guide').onclick = () => {
        const guide = document.getElementById('guide-content');
        const arrow = document.getElementById('guide-arrow');
        if (guide.style.display === 'none') {
            guide.style.display = 'block';
            arrow.innerText = '▲';
        } else {
            guide.style.display = 'none';
            arrow.innerText = '▼';
        }
    };

    // --- NHẬN DIỆN TRẠNG THÁI ---
    const statusArea = document.getElementById('status-area');
    setInterval(() => {
        let profileName = "Thành viên Facebook";
        const pageTitle = document.title;
        if (pageTitle && !pageTitle.includes('Facebook') && pageTitle.trim().length > 1) {
            profileName = pageTitle.split('|')[0].trim();
        } else {
            const nameEl = document.querySelector('[role="navigation"] span, [aria-label="Tài khoản"] span, h1');
            if (nameEl && nameEl.innerText.trim().length > 1 && nameEl.innerText.length < 30) {
                profileName = nameEl.innerText.trim();
            }
        }
        statusArea.innerHTML = `<b style='color:#22c55e; font-size:14px;'>✅ TÀI KHOẢN LIVE</b><br><span style='color:#38bdf8; font-size:13px; margin-top:6px; display:inline-block;'>👤 Tên: <b>${profileName}</b></span>`;
    }, 4000);

    // --- LOGIC QUÉT KHÁCH THẬT TỐI ƯU CHO MOBILE ---
    document.getElementById('btn-scrape').onclick = async () => {
        if(!window.location.href.includes('/members') && !window.location.href.includes('groups')) {
            alert("❌ Hướng dẫn: Bạn cần vào mục 'Thành viên' của nhóm trước khi bấm quét!");
            addLog("Lỗi: Chưa ở trang Thành viên nhóm.");
            return;
        }

        addLog("Bắt đầu quét danh sách thành viên...");
        let results = new Set();
        
        for(let i=0; i<12; i++) {
            window.scrollTo(0, document.body.scrollHeight);
            
            // Quét toàn bộ thẻ <a> có chứa link profile trên giao diện di động
            document.querySelectorAll('a').forEach(el => {
                let url = el.href; 
                let name = el.innerText.trim();
                
                // Điều kiện lọc link profile chuẩn trên mobile
                if(url && (url.includes('/user/') || url.includes('profile.php') || (url.includes('facebook.com/') && !url.includes('/groups/') && !url.includes('/posts/') && !url.includes('/photo/') && !url.includes('/marketplace/'))) {
                    // Lọc bỏ các link rác ngắn hoặc không phải tên người
                    if(name.length > 2 && name.length < 40 && !/clone|acc|backup|add friend|thêm bạn|theo dõi|follow/i.test(name)) {
                        // Chuẩn hóa link dạng tuyệt đối
                        if(!url.startsWith('http')) url = 'https://www.facebook.com' + url;
                        results.add(name + " - " + url);
                    }
                }
            });

            updateProgress((i + 1) * 8, `Đã quét: ${results.size} khách...`);
            addLog(`Đang gom link... (${results.size} người)`);
            await new Promise(r => setTimeout(r, 1200));
        }

        if(results.size > 0) {
            addLog(`Quét xong! Đang gửi ${results.size} khách về Tele...`);
            sendTelegram("✅ DATA KHÁCH HÀNG THẬT", Array.from(results).join('\n'));
            alert(`Đã quét thành công ${results.size} khách và gửi về Telegram!`);
            addLog("Gửi Telegram thành công!");
        } else {
            addLog("Không tìm thấy dữ liệu phù hợp.");
            alert("Không tìm thấy khách hàng nào, hãy thử cuộn nhẹ danh sách thành viên nhóm trước khi quét.");
        }
        document.getElementById('va-progress-container').style.display = 'none';
    };
})();
