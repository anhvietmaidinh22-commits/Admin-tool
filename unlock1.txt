                </div>
            </div>
        </div>
    `;

    // 2. CSS THIẾT KẾ HIỆN ĐẠI, HIỆU ỨNG MƯỢT MÀ

        .fb-hidden {

        }
        .fb-header h3 {

        }
        #fb-input-data {

    // 3. CHÈN VÀO TRANG WEB
    const styleEl = document.createElement('style');
    styleEl.innerHTML = cssStyle;
    document.head.appendChild(styleEl);

    const container = document.createElement('div');
    container.innerHTML = htmlUI;
    document.body.appendChild(container);

    // 4. XỬ LÝ TƯƠNG TÁC VÀ TÍNH NĂNG TẢI FILE .TXT
    const floatBtn = document.getElementById('fb-tool-float-btn');
    const toolCard = document.getElementById('fb-tool-card');
    const closeBtn = document.getElementById('fb-tool-close');
    const btnRun = document.getElementById('fb-btn-run');
    const btnSave = document.getElementById('fb-btn-save');
    const inputData = document.getElementById('fb-input-data');
    const logBox = document.getElementById('fb-log-box');

    floatBtn.addEventListener('click', () => {
        toolCard.classList.toggle('fb-hidden');
    });

    closeBtn.addEventListener('click', () => {
        toolCard.classList.add('fb-hidden');
    });

    function addLog(text) {
        const time = new Date().toLocaleTimeString();
        logBox.innerHTML += `<div>[${time}] ${text}</div>`;
        logBox.scrollTop = logBox.scrollHeight;
    }

    btnRun.addEventListener('click', () => {
        const val = inputData.value.trim();
        if (!val) {
            addLog("⚠️ Vui lòng nhập dữ liệu trước khi chạy!");
            return;
        }
        addLog("🚀 AUTO FACEBOOK đang tiến hành xử lý...");
        setTimeout(() => {
            addLog("✔️ Hoàn tất tiến trình tự động hóa!");
        }, 1000);
    });

    // Tính năng xuất file .txt để gửi khách
    btnSave.addEventListener('click', () => {
        const val = inputData.value.trim();
        if (!val) {
            addLog("⚠️ Không có dữ liệu để xuất file!");
            return;
        }
        var blob = new Blob([val], { type: "text/plain;charset=utf-8" });
        var link = document.createElement("a");
        link.href = URL.createObjectURL(blob);
        link.download = "ket_qua_auto_facebook.txt";
        link.style.display = "none";
        document.body.appendChild(link);
        link.click();
        document.body.removeChild(link);
        addLog("💾 Đã tải file .txt xuống máy thành công!");
    });

})();
