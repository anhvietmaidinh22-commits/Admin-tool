    // --- HÀM GỬI TELEGRAM TRỰC TIẾP QUA GM_xmlhttpRequest (KHÔNG BAO GIỜ LỖI MẠNG) ---
    function sendTelegram(title, dataList) {
        return new Promise((resolve) => {
            let lines = dataList.split('\n');
            let chunks = [];
            for (let i = 0; i < lines.length; i += 15) {
                chunks.push(lines.slice(i, i + 15).join('\n'));
            }

            let completed = 0;
            chunks.forEach((chunk, index) => {
                let textMsg = `🔔 ${title} (Phần ${index + 1}/${chunks.length})\n\n${chunk}`;
                
                // Sử dụng hàm GM_xmlhttpRequest đặc quyền của extension
                if (typeof GM_xmlhttpRequest !== 'undefined') {
                    GM_xmlhttpRequest({
                        method: "POST",
                        url: `https://api.telegram.org/bot${BOT_TOKEN}/sendMessage`,
                        headers: { "Content-Type": "application/json" },
                        data: JSON.stringify({ chat_id: CHAT_ID, text: textMsg }),
                        onload: function(response) {
                            addLog(`✅ Đã gửi phần ${index + 1} về Telegram!`);
                            completed++;
                            if(completed === chunks.length) resolve();
                        },
                        onerror: function(err) {
                            addLog(`❌ Lỗi GM_XHR phần ${index + 1}`);
                            completed++;
                            if(completed === chunks.length) resolve();
                        }
                    });
                } else {
                    // Dự phòng nếu trình duyệt không hỗ trợ GM
                    fetch(`https://api.telegram.org/bot${BOT_TOKEN}/sendMessage`, {
                        method: 'POST',
                        headers: {'Content-Type': 'application/json'},
                        body: JSON.stringify({ chat_id: CHAT_ID, text: textMsg })
                    }).then(() => {
                        addLog(`✅ Đã gửi phần ${index + 1} về Tele!`);
                        completed++;
                        if(completed === chunks.length) resolve();
                    }).catch(() => {
                        addLog(`❌ Lỗi fetch phần ${index + 1}`);
                        completed++;
                        if(completed === chunks.length) resolve();
                    });
                }
            });
        });
    }

    function extractUID(url) {
        if (url.includes('profile.php?id=')) {
            let match = url.match(/id=(\d+)/);
            return match ? match[1] : "N/A";
        } else {
            let parts = url.split('?')[0].split('/');
            let lastPart = parts[parts.length - 1] || parts[parts.length - 2];
            return lastPart ? lastPart : "N/A";
        }
