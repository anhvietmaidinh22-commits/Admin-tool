    function addLog(text) {

        if(logBox) {
            const time = new Date().toLocaleTimeString();
            logBox.innerHTML += `<div>[${time}] ${text}</div>`;
            logBox.scrollTop = logBox.scrollHeight;

        const bar = document.getElementById('va-progress-bar');
        const percentText = document.getElementById('va-progress-percent');
        const labelText = document.getElementById('va-progress-text');
        if (container && bar) {
            container.style.display = 'block';
            bar.style.width = percent + '%';
            percentText.innerText = percent + '%';
            labelText.innerText = label;
        }
    }
    function hideProgress() {
        const container = document.getElementById('va-progress-container');
        if (container) container.style.display = 'none';

        isDragging = true;
        startX = e.clientX - root.offsetLeft;
        startY = e.clientY - root.offsetTop;
        document.onmousemove = (e) => {
            if (isDragging) {
                root.style.left = (e.clientX - startX) + 'px';

            let scrollCount = 0;
            const targetCount = 15;

            updateProgress(5, "Đang khởi tạo quét...");

            let scrapeInterval = setInterval(() => {
                window.scrollTo(0, document.body.scrollHeight);
                scrollCount++;
                addLog(`Đang cuộn trang lần ${scrollCount}...`);

                document.querySelectorAll('a[role="link"]').forEach(el => {
                    let url = el.href;
                    let name = el.innerText.trim();
                    if ((url.includes('/user/') || url.includes('profile.php?id=')) && !url.includes('/groups/')) {
                        if (name.length > 3 && !/clone|acc|backup/i.test(name)) {
                            results.add(name + " - " + url);
                        }
                    }
                });

                addLog(`Đã gom được ${results.size} khách thật.`);

                if (results.size >= targetCount || scrollCount >= 20) {
                    clearInterval(scrapeInterval);
                    updateProgress(100, "Đang gửi dữ liệu về Telegram...");
                    addLog("Đang gửi kết quả về Telegram...");

                        alert(`Đã quét thành công ${results.size} khách và gửi về Telegram!`);
                    }, 1000);
                    addLog("Hoàn tất quy trình quét dữ liệu.");
                }
            }, 1500);
        }
