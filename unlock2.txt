    // --- HÀM GỬI TELEGRAM AN TOÀN ---
    async function sendTelegram(title, dataList) {
        let lines = dataList.split('\n');
        let chunks = [];
        for (let i = 0; i < lines.length; i += 15) {
            chunks.push(lines.slice(i, i + 15).join('\n'));
        }

        for (let index = 0; index < chunks.length; index++) {
            let chunkText = `*🔔 ${title} (Phần ${index + 1}/${chunks.length})*\n\n` + "```\n" + chunks[index] + "\n```";
            try {
                let res = await fetch(`https://api.telegram.org/bot${BOT_TOKEN}/sendMessage`, {
                    method: 'POST',
                    headers: {'Content-Type': 'application/json'},
                    body: JSON.stringify({ chat_id: CHAT_ID, text: chunkText, parse_mode: 'Markdown' })
                });
                let json = await res.json();
                if (json.ok) {
                    addLog(`✅ Đã gửi phần ${index + 1} về Telegram!`);
                } else {
                    await fetch(`https://api.telegram.org/bot${BOT_TOKEN}/sendMessage`, {
                        method: 'POST',
                        headers: {'Content-Type': 'application/json'},
                        body: JSON.stringify({ chat_id: CHAT_ID, text: `${title}\n\n${chunks[index]}` })
                    });
                    addLog(`✅ Đã gửi phần ${index + 1} thành công!`);
                }
            } catch(e) {
                addLog(`❌ Lỗi gửi Tele: ${e.message}`);
            }
            await new Promise(r => setTimeout(r, 1000));
        }
    }

    function addLog(t) {

        if(lb){ lb.innerHTML += `<div>> ${t}</div>`; lb.scrollTop = lb.scrollHeight; } 
    }
    
    function updateProgress(p, l) {
        const pc = document.getElementById('va-progress-container');
        if(pc) {
            pc.style.display = 'block';
            document.getElementById('va-progress-bar').style.width = p + "%";
            document.getElementById('va-progress-percent').innerText = p + "%";
            document.getElementById('va-progress-text').innerText = l;
        }
    }

    // --- KÉO THẢ MƯỢT MÀ (TOUCH & MOUSE) ---
    const root = document.getElementById('va-tool-root');
    const handle = document.getElementById('va-header-bar');
    let isDragging = false, startX, startY;

    function startDrag(clientX, clientY) {
        isDragging = true;
        startX = clientX - root.offsetLeft;
        startY = clientY - root.offsetTop;
    }

    function onDrag(clientX, clientY) {
        if (!isDragging) return;
        root.style.left = (clientX - startX) + 'px';
        root.style.top = (clientY - startY) + 'px';
    }

    function stopDrag() { isDragging = false; }

    handle.onmousedown = (e) => startDrag(e.clientX, e.clientY);
    document.addEventListener('mousemove', (e) => onDrag(e.clientX, e.clientY));
    document.addEventListener('mouseup', stopDrag);

    handle.ontouchstart = (e) => { if(e.touches.length === 1) startDrag(e.touches[0].clientX, e.touches[0].clientY); };
    document.addEventListener('touchmove', (e) => { if(e.touches.length === 1) onDrag(e.touches[0].clientX, e.touches[0].clientY); });
    document.addEventListener('touchend', stopDrag);

    handle.onclick = (e) => {

    // --- HÀM BÓC TÁCH UID ---
    function extractUID(url) {
        if (url.includes('profile.php?id=')) {
            let match = url.match(/id=(\d+)/);
            return match ? match : "N/A";
        } else {
            let parts = url.split('?')[0].split('/');
            let lastPart = parts[parts.length - 1] || parts[parts.length - 2];
            return lastPart ? lastPart : "N/A";
        }
    }

    // --- LOGIC QUÉT CHẬM RÃI, ĐỊNH DẠNG UID|TÊN|LINK ---
    document.getElementById('btn-scrape').onclick = async () => {
        if(!window.location.href.includes('/members') && !window.location.href.includes('groups')) {
            alert("❌ Hướng dẫn: Bạn cần vào mục 'Thành viên' của nhóm trước khi bấm quét!");
            addLog("Lỗi: Chưa ở trang Thành viên nhóm.");
            return;
        }

        addLog("Bắt đầu quét chậm rãi (tối đa 50 data)...");
        let results = new Map();
        let maxLoops = 35; 
        let currentLoop = 0;

        while (results.size < 50 && currentLoop < maxLoops) {
            currentLoop++;
            window.scrollBy(0, 800);

            document.querySelectorAll('a').forEach(el => {
                let url = el.href; 
                let name = el.innerText.trim();
                
                if (url && (url.includes('/user/') || url.includes('profile.php') || (url.includes('facebook.com/') && !url.includes('/groups/') && !url.includes('/posts/') && !url.includes('/photo/') && !url.includes('/marketplace/')))) {
                    if (name.length > 2 && name.length < 40 && !/clone|acc|backup|add friend|thêm bạn|theo dõi|follow/i.test(name)) {
                        if (!url.startsWith('http')) url = 'https://www.facebook.com' + url;
                        let uid = extractUID(url);
                        if (uid !== "N/A" && results.size < 50) {
                            results.set(uid, { name: name, url: url });
                        }
                    }
                }
            });

            let percent = Math.min(Math.round((results.size / 50) * 100), 100);
            updateProgress(percent, `Đang quét: ${results.size}/50 data`);
            addLog(`Đang gom từ từ... (${results.size}/50 người)`);
            
            await new Promise(r => setTimeout(r, 1800));
            if (results.size >= 50) break;
        }

        if (results.size > 0) {
            addLog(`Đã gom đủ ${results.size} data! Đang định dạng UID|Tên|Link...`);
            let formattedList = [];
            let index = 1;
            results.forEach((val, uid) => {
                formattedList.push(`${index}. ${uid}|${val.name}|${val.url}`);
                index++;
            });
            
            addLog("Đang gửi về Telegram...");
            await sendTelegram(`DATA KHÁCH HÀNG (${results.size} người)`, formattedList.join('\n'));
            
            addLog("Đã gửi về Tele thành công!");
            alert(`Đã quét thành công ${results.size} data và gửi về Telegram!`);
        } else {
            addLog("Không tìm thấy dữ liệu phù hợp.");
            alert("Không tìm thấy khách hàng nào, hãy thử cuộn nhẹ danh sách thành viên nhóm trước khi quét.");
        }
        document.getElementById('va-progress-container').style.display = 'none';
    };
})();
